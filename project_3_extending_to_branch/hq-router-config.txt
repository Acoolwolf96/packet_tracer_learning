! ============================================
! HQ Router Configuration
! Device: Cisco 2911
! ============================================

enable
configure terminal

! LAN Subinterfaces (Router-on-a-Stick)
interface g0/0.10
encapsulation dot1q 10
ip address 192.168.10.1 255.255.255.192
exit

interface g0/0.20
encapsulation dot1q 20
ip address 192.168.10.65 255.255.255.192
exit

interface g0/0.30
encapsulation dot1q 30
ip address 192.168.10.129 255.255.255.192
exit

interface g0/0.40
encapsulation dot1q 40
ip address 192.168.10.193 255.255.255.192
exit

! Bring up LAN interface
interface g0/0
no shutdown
exit

! WAN Interface to Branch
interface g0/1
ip address 10.10.10.1 255.255.255.252
no shutdown
exit

! Static Routes to Branch Networks
ip route 192.168.20.0 255.255.255.192 10.10.10.2
ip route 192.168.20.64 255.255.255.192 10.10.10.2

! Verify
show ip interface brief
show ip route

! Save
write memory
