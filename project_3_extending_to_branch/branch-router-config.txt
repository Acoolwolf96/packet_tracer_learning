! ============================================
! Branch Router Configuration
! Device: Cisco 2911
! ============================================

enable
configure terminal

! LAN Subinterfaces (Router-on-a-Stick)
interface g0/0.10
encapsulation dot1q 10
ip address 192.168.20.1 255.255.255.192
exit

interface g0/0.20
encapsulation dot1q 20
ip address 192.168.20.65 255.255.255.192
exit

! Bring up LAN interface
interface g0/0
no shutdown
exit

! WAN Interface to HQ
interface g0/1
ip address 10.10.10.2 255.255.255.252
no shutdown
exit

! Static Routes to HQ Networks
ip route 192.168.10.0 255.255.255.192 10.10.10.1
ip route 192.168.10.64 255.255.255.192 10.10.10.1
ip route 192.168.10.128 255.255.255.192 10.10.10.1
ip route 192.168.10.192 255.255.255.192 10.10.10.1

! Verify
show ip interface brief
show ip route

! Save
write memory
